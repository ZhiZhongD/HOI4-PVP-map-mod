on_actions = {
    on_weekly_GER = {
        effect = {
			set_temp_variable = {
				var = State_of_Wehrmacht_war_support_temp
				value = root.has_war_support
			}
			add_to_temp_variable = {
				var = State_of_Wehrmacht_war_support_temp
				value = -0.5
			}
			set_variable = {
				var = State_of_Wehrmacht_var_0
				value = State_of_Wehrmacht_war_support_temp
			}
			set_variable = {
				var = State_of_Wehrmacht_var_1
				value = State_of_Wehrmacht_war_support_temp
			}
			multiply_variable = {
				var = State_of_Wehrmacht_var_0
				value = 0.5
			}
			multiply_variable = {
				var = State_of_Wehrmacht_var_1
				value = 0.4
			}

        }
    }
	on_startup = {
		effect = {
			GR1 = {
				every_core_state = {
					add_core_of = GER
				}
			}
			GR4 = {
				every_core_state = {
					add_core_of = GER
				}
			}
			GER = {
				every_core_state = {
					add_core_of = GR1
					add_core_of = GR3
					add_core_of = GR4
					add_core_of = GR5
					add_core_of = GR6
				}
			}
			every_state = {
				limit = {
					is_core_of  = ITA 
				}
				add_core_of = GR5
			}
			every_state = {
				limit = {
					OR = {
						is_core_of  = LIT
						is_core_of  = LAT
						is_core_of  = EST
					}
				}
				add_core_of = GR6
			}
			GR6 = {
				add_manpower = 150000
			}

			#调整开局工厂
				every_state = {
					limit = {
						OR = {
							has_state_category = megalopolis
							has_state_category = metropolis
							has_state_category = large_city
							has_state_category = city
							has_state_category = large_town
							has_state_category = town
						}
					}
					add_extra_state_shared_building_slots = 2
                    add_building_construction = {
                        type = industrial_complex
                        level = 1
                        instant_build = yes
                    }
					add_building_construction = {
                        type = arms_factory
                        level = 1
                        instant_build = yes
                    }
                    add_building_construction = {
                        type = infrastructure
                        level = 1
                        instant_build = yes
                    }
				}
		}
	}
}
